"""Read-only native comparison; writes only this private evidence folder."""
from pathlib import Path
import hashlib
import json
import time

import cadquery as cq

HERE = Path(__file__).resolve().parent
ROOT = Path('/Users/derekbredensteiner/Developer/homesodamachine')
OLD = HERE.parent / 'enclosure-back-top.brep'
NEW = ROOT / 'hardware/printed-parts/enclosure/enclosure/enclosure-back-top.step'
OLD_MANIFEST = HERE.parent / 'current-native.json'
OLD_STEP = HERE.parent / 'before/hardware/manifold-layout/enclosure-assembly.step'

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def bounds(s):
    b = s.BoundingBox()
    return [b.xmin, b.ymin, b.zmin, b.xmax, b.ymax, b.zmax]

def describe(s):
    return dict(valid=s.isValid(), solids=len(s.Solids()), faces=len(s.Faces()),
                volume_mm3=s.Volume(), area_mm2=s.Area(), bounds_mm=bounds(s))

t0 = time.monotonic()
manifest = json.loads(OLD_MANIFEST.read_text())
assert sha(OLD) == manifest['bodies']['enclosure-back-top']['sha256']
assert sha(OLD_STEP) == manifest['source_sha256']
before_hashes = {str(p): sha(p) for p in (OLD, NEW, OLD_MANIFEST, OLD_STEP)}
old = cq.Shape.importBrep(str(OLD))
new = cq.importers.importStep(str(NEW)).val()
report = {'scope': 'Current canonical native back-top minus/against the exact placed leaf extracted from the prior qualified aggregate. No tessellated-surface comparison is substituted for native booleans.',
          'inputs_sha256': before_hashes,
          'source_sha256': {str(Path(__file__)): sha(Path(__file__))},
          'prior': describe(old), 'current': describe(new), 'operations': {}}
(HERE / 'native-summary.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({'prior': report['prior'], 'current': report['current']}, indent=2), flush=True)
for key, a, b in [('added', new, old), ('removed', old, new)]:
    t = time.monotonic()
    print('start', key, flush=True)
    delta = a.cut(b)
    solids = delta.Solids()
    result = dict(volume_mm3=sum(s.Volume() for s in solids), valid=delta.isValid(),
                  solids=len(solids), elapsed_seconds=time.monotonic()-t,
                  components=[describe(s) for s in solids])
    if solids:
        dst = HERE / (key + '.brep')
        delta.exportBrep(str(dst))
        result['native_witness'] = str(dst)
        result['native_witness_sha256'] = sha(dst)
    report['operations'][key] = result
    print(json.dumps({key: result}, indent=2), flush=True)
    (HERE / 'native-summary.json').write_text(json.dumps(report, indent=2) + '\n')
report['inputs_stable'] = all(sha(Path(p)) == h for p, h in before_hashes.items())
report['elapsed_seconds'] = time.monotonic()-t0
report['status'] = ('geometrically_equal_within_native_tolerance'
                    if all(x['volume_mm3'] < 1e-6 for x in report['operations'].values())
                    else 'geometric_difference')
(HERE / 'native-summary.json').write_text(json.dumps(report, indent=2) + '\n')
print('status', report['status'], 'seconds', report['elapsed_seconds'], flush=True)
