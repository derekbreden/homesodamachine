"""Validate a newly completed support ledger; never release or submit a print."""
from pathlib import Path
import json

from print_workflow import HERE, ROOT, PROFILE, sha, rel, resolve, write_json, load_settings, staged_report


def nonempty(value):
    return isinstance(value, str) and bool(value.strip())


def checked_evidence(row, archive_hash, gcode_hash):
    if not isinstance(row, dict) or not row.get('path') or not row.get('sha256'):
        raise ValueError('A named hash-bound current evidence file is required')
    path = resolve(row['path'])
    if sha(path) != row['sha256']:
        raise ValueError(f'Current evidence changed: {path}')
    proof = json.loads(path.read_text())
    if (proof.get('status') != 'pass' or proof.get('native_archive_sha256') != archive_hash
            or proof.get('gcode_sha256') != gcode_hash):
        raise ValueError(f'The evidence does not qualify this exact native archive: {path}')
    return proof


def validate_ledger(readiness, audit, ledger):
    if ledger.get('status') != 'current_support_contacts_and_removal_lanes_reviewed':
        raise ValueError('The exact current support ledger is not marked reviewed')
    expected = {
        'native_archive_sha256': readiness['native_archive_sha256'],
        'gcode_sha256': readiness['gcode_sha256'],
        'support_audit_sha256': readiness['support_audit_sha256'],
        'geometry_confirmation_sha256': readiness['geometry_confirmation_sha256'],
    }
    for key, digest in expected.items():
        if ledger.get(key) != digest:
            raise ValueError(f'The support ledger is stale or incomplete: {key}')
    measured_parts = {r['piece']: r for r in audit['parts']}
    named_parts = [r['name'] for r in ledger['parts']]
    if len(named_parts) != len(set(named_parts)) or set(named_parts) != set(measured_parts):
        raise ValueError('Every current plate object needs exactly one support review')
    for part in ledger['parts']:
        measured = measured_parts[part['name']]
        if part['support_summary'] != measured['summary'] or not nonempty(part.get('model_overhangs_review')):
            raise ValueError(f'Complete the model/support summary review for {part["name"]}')
        for family in ('trees', 'interfaces'):
            actual = {r['id']: r for r in measured[family]}
            rows = part[family]
            ids = [r['measurement']['id'] for r in rows]
            if len(ids) != len(set(ids)) or set(ids) != set(actual):
                raise ValueError(f'Cover every current {family} measurement for {part["name"]}')
            for row in rows:
                if row['measurement'] != actual[row['measurement']['id']]:
                    raise ValueError('A measured support topology row changed')
                if row.get('reviewed') is not True or not nonempty(row.get('contact_region')):
                    raise ValueError('Every support/contact region needs an explicit current review')
                if family == 'trees':
                    required = ('removal_lane', 'removal_direction_original_cad', 'preserved_working_face_reason')
                    if (not all(nonempty(row.get(key)) for key in required)
                            or row.get('accessible_before_hardware') is not True):
                        raise ValueError('Name each detached branch route and accessible assembly state')
    if ledger.get('native_log_diagnostics') != readiness['native_log_diagnostics']:
        raise ValueError('The ledger must name the exact current native log diagnostics')
    if readiness['native_log_diagnostics'] and not nonempty(ledger.get('native_log_diagnostics_review')):
        raise ValueError('Explain each current native diagnostic before finalizing')
    fit = readiness['toolpath_fit']
    if not fit['bounds_inside_nozzle_area']:
        raise ValueError('Emitted roads leave the nozzle area; this plate needs correction')
    if fit['overlap_investigation_needed']:
        proof = checked_evidence(ledger.get('pairwise_path_overlap_evidence'),
                                 readiness['native_archive_sha256'], readiness['gcode_sha256'])
        if proof.get('all_current_pairs_clear') is not True:
            raise ValueError('Projected path overlap needs exact current layerwise clearance evidence')
    passed = set()
    reading = readiness.get('carrier_retaining_wall_toolpath_reading')
    if reading and reading.get('status') == 'pass':
        passed.add(reading['id'])
    if (readiness.get('spring_pusher_toolpath_reading') or {}).get('all_layers_have_connected_head_and_tongue'):
        passed.add('spring-pusher-connectivity')
    for row in ledger.get('additional_functional_toolpath_evidence', []):
        proof = checked_evidence(row, readiness['native_archive_sha256'], readiness['gcode_sha256'])
        passed.update(proof.get('passed_checks', []))
    missing = set(readiness['required_functional_toolpath_checks']) - passed
    if missing:
        raise ValueError(f'Current functional toolpaths remain unchecked: {sorted(missing)}')
    return expected


def main():
    _staged, report = staged_report()
    path = HERE / 'readiness.json'
    readiness = json.loads(path.read_text())
    if readiness['geometry_confirmation_sha256'] != report['geometry_confirmation_sha256']:
        raise ValueError('The native review belongs to another geometry receipt')
    for key, digest in (('native_archive', 'native_archive_sha256'),
                        ('support_audit', 'support_audit_sha256')):
        if sha(resolve(readiness[key])) != readiness[digest]:
            raise ValueError(f'The native review input changed: {key}')
    ledger_path = HERE / 'support-removal-review.json'
    ledger = json.loads(ledger_path.read_text())
    audit = json.loads(resolve(readiness['support_audit']).read_text())
    validate_ledger(readiness, audit, ledger)
    result = {**readiness,
        'status': 'offline_print_review_pass', 'submitted': False, 'print_released': False,
        'final_print_release': 'pending_coordinating_task_printer_handoff_and_release',
        'printer_state_checked': False,
        'support_removal_review_sha256': sha(ledger_path),
        'native_log_diagnostics_review': ledger['native_log_diagnostics_review'],
        'native_log_diagnostics_review_reference': {'path': rel(ledger_path), 'sha256': sha(ledger_path), 'field': 'native_log_diagnostics_review'},
        'native_review_sha256': sha(path), 'final_review_script_sha256': sha(Path(__file__)),
        'brim_reading': {'staged_type': 'auto_brim', 'actual_brim_extrusion_segments':
                        sum(p['toolpaths']['features'].get('Brim', 0) for p in audit['parts'])},
        'estimated_grams_petgf_density_1p43': readiness['estimated_grams_saved_profile_density']
                        / float(load_settings(PROFILE)['filament_density'][0]) * 1.43,
    }
    write_json(HERE / 'offline-review.json', result)
    print(json.dumps({key: result[key] for key in ('status', 'native_archive', 'native_archive_sha256',
                     'estimated_seconds', 'estimated_grams_petgf_density_1p43', 'brim_reading')}, indent=2))


if __name__ == '__main__':
    main()
