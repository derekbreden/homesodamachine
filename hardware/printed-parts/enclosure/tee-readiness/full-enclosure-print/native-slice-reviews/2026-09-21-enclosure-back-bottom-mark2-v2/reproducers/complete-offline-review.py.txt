"""Mirror the completed exact support-ledger diagnostic reading into final metadata."""
from pathlib import Path
import argparse
import hashlib
import json

ROOT = Path('/Users/derekbredensteiner/Developer/homesodamachine')
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('job_directory', type=Path)
    args = parser.parse_args()
    job = args.job_directory.resolve()
    path = job / 'offline-review.json'
    ledger_path = job / 'support-removal-review.json'
    original = path.read_bytes()
    review, ledger = json.loads(original), json.loads(ledger_path.read_text())
    assert review['status'] == 'offline_print_review_pass'
    assert ledger['status'] == 'current_support_contacts_and_removal_lanes_reviewed'
    assert sha(ledger_path) == review['support_removal_review_sha256']
    for key in ('native_archive_sha256', 'gcode_sha256', 'support_audit_sha256',
                'geometry_confirmation_sha256', 'native_log_diagnostics'):
        assert review[key] == ledger[key], key
    completed = ledger['native_log_diagnostics_review']
    if not review['native_log_diagnostics']:
        completed = completed or 'No matching native log diagnostics.'
    assert isinstance(completed, str) and completed.strip() and completed != 'pending'
    if review.get('native_log_diagnostics_review_source'):
        assert review['native_log_diagnostics_review'] == completed
        print(json.dumps({'status': 'already_complete', 'sha256': sha(path)}))
        return
    before = job / 'offline-review-before-diagnostic-mirror-fix.json'
    if before.exists():
        assert before.read_bytes() == original
    else:
        before.write_bytes(original)
    review['native_log_diagnostics_review'] = completed
    review['native_log_diagnostics_review_source'] = {
        'path': str(ledger_path.relative_to(ROOT)), 'sha256': sha(ledger_path),
        'field': 'native_log_diagnostics_review',
    }
    review['metadata_completion'] = {
        'predecessor_path': str(before.relative_to(ROOT)), 'predecessor_sha256': sha(before),
        'script_path': str(Path(__file__).resolve()), 'script_sha256': sha(__file__),
        'scope': 'Mirror the completed bound-ledger explanation. Native archive, '
                 'G-code, settings, geometry, measurements and release state are unchanged.',
    }
    path.write_text(json.dumps(review, indent=2) + '\n')
    print(json.dumps({'status': review['status'], 'sha256': sha(path),
                      'diagnostic_review': 'completed_bound_ledger_explanation',
                      'native_archive_sha256': review['native_archive_sha256']}, indent=2))


if __name__ == '__main__':
    main()
