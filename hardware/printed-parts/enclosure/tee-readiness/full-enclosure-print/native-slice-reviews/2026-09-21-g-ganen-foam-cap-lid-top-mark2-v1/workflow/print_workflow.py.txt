"""Private, offline production print preparation from an exact fresh receipt.

Preparation, native slicing and review are separate explicit commands. None
connects to a printer, changes production geometry or marks a print released.
"""
from pathlib import Path
import argparse
import hashlib
import json
import os
import re
import subprocess
import sys
import zipfile

import numpy as np
import trimesh

HERE = Path(__file__).resolve().parent
ROOT = next(p for p in HERE.parents if (p / 'hardware/printed-parts/petgf.3mf').is_file())
PROFILE = ROOT / 'hardware/printed-parts/petgf.3mf'
PROFILE_SHA = '3864deceaa295d0fffded05e445b77ee926e2706d8765fdc8bbf81a23fe98697'
BOX = ROOT / 'hardware/manifold-layout/enclosure-box.json'
AREA = np.array([[0., 0.], [325., 320.]])
IDENTITY = np.eye(3, dtype=int)
sys.path.insert(0, str(ROOT / 'hardware/printed-parts/faucet'))
import refresh_print_project as writer


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def rel(path):
    path = Path(path).resolve()
    return str(path.relative_to(ROOT)) if path.is_relative_to(ROOT) else str(path)


def resolve(path):
    return (ROOT / path).resolve()


def write_json(path, data):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2) + '\n')


def load_settings(path):
    with zipfile.ZipFile(path) as archive:
        return json.loads(archive.read(writer.SETTINGS_MEMBER))


def configuration():
    config = json.loads((HERE / 'job-config.json').read_text())
    if config['printer'] not in ('H2C', 'Mark2'):
        raise ValueError('Unknown printer')
    trim = .18 if config['printer'] == 'H2C' else .04
    if config['requested_z_trim_mm'] != trim or config['model_border_mm'] != 15.:
        raise ValueError('Unexpected printer trim or established model border')
    if not re.fullmatch(r'[a-z0-9-]+', config['archive_stem']):
        raise ValueError('Archive stem must be a private simple filename')
    if not config['parts'] or len({p['name'] for p in config['parts']}) != len(config['parts']):
        raise ValueError('Parts must be nonempty and uniquely named')
    return config


def parts_for(config):
    return [(p['name'], resolve(p['stl']), resolve(p['step']),
             np.asarray(p['rotation_matrix'], dtype=int), p['rotation_sequence'])
            for p in config['parts']]


def verify_confirmation(path, parts, required_sources=()):
    """No source-equivalence exceptions: all declared inputs must match now."""
    path = Path(path)
    receipt = json.loads(path.read_text())
    if (receipt.get('status') != 'current_geometry_confirmed'
            or not isinstance(receipt.get('authority'), str)
            or not receipt['authority'].strip()
            or receipt.get('print_released') is not False):
        raise ValueError('An explicit fresh geometry confirmation is required before preparation')
    maps = {}
    for group in ('artifact_sha256', 'source_sha256', 'evidence_sha256'):
        entries = receipt.get(group)
        if not isinstance(entries, dict) or not entries:
            raise ValueError(f'The geometry confirmation requires a nonempty {group}')
        maps[group] = {rel(resolve(name)): expected for name, expected in entries.items()}
        for name, expected in entries.items():
            if not isinstance(expected, str) or not re.fullmatch(r'[0-9a-f]{64}', expected):
                raise ValueError(f'Invalid SHA256 in {group}: {name}')
            if sha(resolve(name)) != expected:
                raise ValueError(f'Confirmed {group} changed: {name}')
    for _name, mesh, native, _matrix, _sequence in parts:
        for source in (mesh, native):
            if maps['artifact_sha256'].get(rel(source)) != sha(source):
                raise ValueError(f'The fresh artifact receipt omits {rel(source)}')
    if maps['artifact_sha256'].get(rel(BOX)) != sha(BOX):
        raise ValueError('The fresh serialized Box must be bound as an artifact')
    if sha(PROFILE) != PROFILE_SHA or not any(m.get(rel(PROFILE)) == PROFILE_SHA for m in maps.values()):
        raise ValueError('The exact saved PET-GF profile must be hash-bound in the fresh receipt')
    for source in required_sources:
        if maps['source_sha256'].get(rel(resolve(source))) != sha(resolve(source)):
            raise ValueError(f'The fresh source receipt omits {source}')
    return receipt


def workflow_hashes():
    return {rel(p): sha(p) for p in sorted(HERE.iterdir())
            if p.suffix == '.py' or p.name in ('job-config.json', 'layout.json')}


def verify_report(report):
    receipt_path = resolve(report['geometry_confirmation'])
    if sha(receipt_path) != report['geometry_confirmation_sha256']:
        raise ValueError('The bound geometry confirmation changed')
    config = configuration()
    verify_confirmation(receipt_path, parts_for(config), config['required_sources'])
    if sha(Path(writer.__file__)) != report['writer_source_sha256']:
        raise ValueError('The staged shared print-project writer changed')
    for path, digest in report['workflow_sha256'].items():
        if sha(resolve(path)) != digest:
            raise ValueError(f'The staged private workflow changed: {path}')
    if workflow_hashes() != report['workflow_sha256']:
        raise ValueError('The staged private workflow file set changed')
    for original in report['original_sources']:
        for path_key, digest_key in (('source', 'source_stl_sha256'),
                                     ('source_step', 'source_step_sha256'),
                                     ('oriented_source', 'oriented_stl_sha256')):
            if sha(resolve(original[path_key])) != original[digest_key]:
                raise ValueError(f'Staged geometry changed: {original[path_key]}')
    if report.get('functional_regions'):
        declared = resolve(report['functional_regions'])
        if sha(declared) != report['functional_regions_sha256']:
            raise ValueError('The current functional-region declaration changed')
    return receipt_path


def orient(parts):
    records, meshes = [], {}
    cache = HERE / 'oriented-meshes'
    cache.mkdir(exist_ok=False)
    for name, path, native, rotation, sequence in parts:
        if (rotation.shape != (3, 3) or not np.array_equal(rotation.T @ rotation, IDENTITY)
                or round(np.linalg.det(rotation)) != 1):
            raise ValueError(f'{name} requires a proper axis permutation')
        source_hash, native_hash = sha(path), sha(native)
        mesh = trimesh.load(path, force='mesh', process=True)
        if not (mesh.is_watertight and mesh.is_winding_consistent
                and mesh.volume > 0 and mesh.body_count == 1):
            raise ValueError(f'{name} is not one closed oriented volume')
        oriented = mesh.copy()
        oriented.vertices = oriented.vertices @ rotation.T
        target = cache / (name + '.stl')
        oriented.export(target)
        reloaded = trimesh.load(target, force='mesh', process=True)
        recovered = np.asarray(sorted(map(tuple, reloaded.vertices @ rotation)))
        original = np.asarray(sorted(map(tuple, mesh.vertices)))
        if original.shape != recovered.shape or not np.array_equal(original, recovered):
            raise ValueError(f'{name} rigid transform changed the source vertices')
        if sha(path) != source_hash or sha(native) != native_hash:
            raise ValueError(f'{name} changed during orientation')
        meshes[name] = reloaded
        records.append({
            'name': name, 'source': rel(path), 'source_stl_sha256': source_hash,
            'source_step': rel(native), 'source_step_sha256': native_hash,
            'original_bounds_mm': mesh.bounds.tolist(), 'original_mesh_triangles': len(mesh.faces),
            'pretransform_rotation_sequence': sequence,
            'pretransform_rotation_matrix': rotation.tolist(),
            'pretransform_translation_mm': [0., 0., 0.],
            'oriented_source': rel(target), 'oriented_stl_sha256': sha(target),
            'oriented_bounds_mm': reloaded.bounds.tolist(),
            'rigid_transform_vertex_recovery_exact': True,
            'print_up_in_original_cad': rotation[2].tolist(),
        })
    return records, meshes


def packing(config, meshes):
    if config['layout'] == 'explicit':
        rows = json.loads((HERE / 'layout.json').read_text())
        centers = {k: np.asarray(v, dtype=float) for k, v in rows.items()}
    elif config['layout'] == 'centered-single':
        if len(meshes) != 1:
            raise ValueError('A centered single plate must have one part')
        centers = {name: AREA.mean(axis=0) for name in meshes}
    elif config['layout'] == 'centered-carrier-pair':
        left, right = 'enclosure-tee-carrier-left', 'enclosure-tee-carrier-right'
        if set(meshes) != {left, right}:
            raise ValueError('The carrier plate requires both complete halves')
        widths = {name: mesh.extents[:2] for name, mesh in meshes.items()}
        gap = config['model_pair_gap_mm']
        start = (AREA[1, 0] - widths[left][0] - gap - widths[right][0]) / 2
        centers = {
            left: np.array([start + widths[left][0] / 2, 160.]),
            right: np.array([start + widths[left][0] + gap + widths[right][0] / 2, 160.]),
        }
    else:
        raise ValueError('Unknown layout')
    if centers.keys() != meshes.keys():
        raise ValueError('The layout must name exactly the current plate objects')
    bounds = {}
    for name, center in centers.items():
        if center.shape != (2,) or not np.all(np.isfinite(center)):
            raise ValueError(f'Invalid placement for {name}')
        half = meshes[name].extents[:2] / 2
        low, high = center - half, center + half
        if np.any(low < config['model_border_mm']) or np.any(high > AREA[1] - config['model_border_mm']):
            raise ValueError(f'{name} does not retain the 15 mm model border: {low}, {high}')
        bounds[name] = (low, high)
    for i, name in enumerate(bounds):
        low, high = bounds[name]
        for other in list(bounds)[:i]:
            other_low, other_high = bounds[other]
            gap = float(np.linalg.norm(np.maximum(0., np.maximum(low - other_high, other_low - high))))
            if gap <= 0.:
                raise ValueError(f'Model footprint overlap: {name}, {other}')
    return centers


def prepare(confirmation, functional_regions=None):
    config = configuration()
    parts = parts_for(config)
    receipt = verify_confirmation(confirmation, parts, config['required_sources'])
    receipt_hash = sha(confirmation)
    name = config['archive_stem']
    staged = HERE / (name + '-input.3mf')
    if any(p.exists() for p in (staged, HERE / 'preparation.json', HERE / 'oriented-meshes',
                                HERE / 'geometry-confirmation.json', HERE / 'ready')):
        raise ValueError('This job already has staged work; keep it immutable and use a new version')
    if functional_regions:
        if receipt['evidence_sha256'].get(rel(functional_regions)) != sha(functional_regions):
            raise ValueError('The functional-region declaration must be in the fresh evidence map')
    records, meshes = orient(parts)
    centers = packing(config, meshes)
    trim = config['requested_z_trim_mm']
    report = writer.refresh(
        PROFILE, staged,
        parts=tuple((r['name'], resolve(r['oriented_source']), 0.) for r in records),
        offsets=tuple(tuple(centers[r['name']] - AREA.mean(axis=0)) for r in records),
        title=config['title'], z_trim=trim, plate_border=config['model_border_mm'],
    )
    if report['shared_printable_area_mm'] != AREA.tolist():
        raise ValueError('The saved nozzle area differs from 325 x 320 mm')
    with zipfile.ZipFile(staged) as archive:
        members = {member: archive.read(member) for member in archive.namelist()}
    settings = json.loads(members[writer.SETTINGS_MEMBER])
    settings['extruder_ams_count'] = ['1#0|4#0', '1#0|4#0']
    if settings['brim_type'] != 'auto_brim':
        raise ValueError('New production plates must inherit the saved auto_brim setting')
    if settings['print_sequence'] != 'by layer':
        raise ValueError('The plate must print by whole layer')
    if settings['filament_nozzle_map'] != ['0'] or settings['filament_colour'] != ['#000000']:
        raise ValueError('The plate requires black PET-GF on the left nozzle')
    members[writer.SETTINGS_MEMBER] = (json.dumps(settings, indent=2) + '\n').encode()
    writer.archive_write(staged, members)
    local_receipt = HERE / 'geometry-confirmation.json'
    local_receipt.write_bytes(Path(confirmation).read_bytes())
    report.update({
        'project_sha256': sha(staged),
        'settings_sha256': hashlib.sha256(members[writer.SETTINGS_MEMBER]).hexdigest(),
        'job_key': config['job_key'], 'assigned_printer': config['printer'],
        'archive_stem': name, 'requested_z_trim_mm': trim,
        'actual_textured_z_trim_mm': round(trim - .02, 2),
        'geometry_confirmation': rel(local_receipt), 'geometry_confirmation_sha256': receipt_hash,
        'geometry_confirmation_received_from': rel(confirmation),
        'original_sources': records, 'layerwise_print': True,
        'print_released': False, 'final_print_release': 'pending_coordinating_task',
        'writer_source_sha256': sha(Path(writer.__file__)), 'workflow_sha256': workflow_hashes(),
        'functional_regions': rel(functional_regions) if functional_regions else None,
        'functional_regions_sha256': sha(functional_regions) if functional_regions else None,
        'required_functional_toolpath_checks': config.get('required_functional_toolpath_checks', []),
    })
    for part, original in zip(report['parts'], records):
        part['original_source'] = original['source']
        part['original_stl_sha256'] = original['source_stl_sha256']
    verify_confirmation(confirmation, parts, config['required_sources'])
    if sha(confirmation) != receipt_hash or sha(local_receipt) != receipt_hash:
        raise ValueError('The geometry confirmation changed during preparation')
    write_json(staged.with_suffix('.print.json'), report)
    write_json(HERE / 'preparation.json', {
        'status': 'current_geometry_staged_native_slice_pending', 'submitted': False,
        'print_released': False, 'staged_input': rel(staged), 'staged_input_sha256': sha(staged),
        'printer': config['printer'], 'parts': records,
        'centers_xy_mm': {k: v.tolist() for k, v in centers.items()},
        'profile_sha256': PROFILE_SHA, 'geometry_confirmation_sha256': receipt_hash,
    })
    return staged, report


def staged_report():
    prep = json.loads((HERE / 'preparation.json').read_text())
    staged = resolve(prep['staged_input'])
    if sha(staged) != prep['staged_input_sha256']:
        raise ValueError('The staged input changed')
    report = json.loads(staged.with_suffix('.print.json').read_text())
    if sha(staged) != report['project_sha256']:
        raise ValueError('The staged report does not bind this input')
    verify_report(report)
    return staged, report


def slice_native():
    staged, report = staged_report()
    ready = HERE / 'ready'
    if ready.exists():
        raise ValueError('A native attempt already exists; preserve it and use a new job version')
    ready.mkdir()
    archive = ready / (report['archive_stem'] + '.gcode.3mf')
    command = ['/Applications/BambuStudio.app/Contents/MacOS/BambuStudio',
               '--slice', '0', '--arrange', '0', '--orient', '0',
               '--outputdir', str(ready), '--export-3mf', archive.name, str(staged)]
    write_json(HERE / 'slice-command.json', command)
    with (ready / 'bambu-cli.log').open('w') as log:
        process = subprocess.run(command, cwd=ready, stdout=log, stderr=subprocess.STDOUT,
                                 env={**os.environ, 'OMP_NUM_THREADS': '1', 'OPENBLAS_NUM_THREADS': '1'})
    if process.returncode:
        raise RuntimeError(f'Native slice failed: {ready / "bambu-cli.log"}')
    verify_report(report)
    write_json(HERE / 'slice-input.json', {
        'input_sha256': sha(staged), 'archive_sha256': sha(archive),
        'geometry_confirmation_sha256': report['geometry_confirmation_sha256'],
    })
    print(json.dumps({'native_archive': rel(archive), 'sha256': sha(archive),
                      'status': 'native_archive_exists_review_pending'}, indent=2), flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--geometry-confirmation', type=Path, required=True)
    parser.add_argument('--functional-regions', type=Path,
                        help='Fresh receipt-bound native region declaration for emitted wall checks')
    args = parser.parse_args()
    staged, report = prepare(args.geometry_confirmation.resolve(),
                             args.functional_regions.resolve() if args.functional_regions else None)
    print(json.dumps({'staged': rel(staged), 'sha256': sha(staged),
                      'parts': [{k: p[k] for k in ('name', 'plate_bounds_mm')} for p in report['parts']]}, indent=2))


if __name__ == '__main__':
    main()
